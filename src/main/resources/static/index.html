Math.random() * sCanvas.height : -35; this.size = 16 + Math.random() * 20; // Crisp flower size (16px - 36px) this.speedY = 0.8 + Math.random() * 1.5; this.speedX = -0.4 + Math.random() * 1.0; this.oscillationSpeed = 0.015 + Math.random() * 0.025; this.oscillationPhase = Math.random() * Math.PI * 2; this.angle = Math.random() * Math.PI * 2; this.spinSpeed = (Math.random() - 0.5) * 0.035; this.wobble = Math.random() * Math.PI * 2; this.wobbleSpeed = 0.02 + Math.random() * 0.035; this.opacity = 0.55 + Math.random() * 0.42; this.depthScale = 0.65 + Math.random() * 0.55; this.imgIndex = Math.random() > 0.45 ? 0 : 1; } update() { this.y += this.speedY; this.angle += this.spinSpeed; this.wobble += this.wobbleSpeed; this.x += this.speedX + Math.sin(this.y * this.oscillationSpeed + this.oscillationPhase) * 0.75; if (this.y > sCanvas.height + 40 || this.x > sCanvas.width + 45 || this.x < -45) this.reset(false); } draw(ctx) { const img = flowerImgs[this.imgIndex]; if (!img.complete || img.naturalWidth === 0) return; ctx.save(); ctx.translate(this.x, this.y); ctx.rotate(this.angle); // Realistic 3D fluttering in the breeze (gentle squish along X-axis as it tumbles) const flutterScaleX = Math.cos(this.wobble) * 0.9; ctx.scale(this.depthScale * flutterScaleX, this.depthScale); ctx.globalAlpha = this.opacity; const s = this.size; ctx.drawImage(img, -s / 2, -s / 2, s, s); ctx.restore(); } } for (let i = 0; i < petalsCount; i++) { petals.push(new SakuraPetal()); } function renderSakura() { sCtx.clearRect(0, 0, sCanvas.width, sCanvas.height); for (const p of petals) { p.update(); p.draw(sCtx); } requestAnimationFrame(renderSakura); } renderSakura(); // ------------------------------------------------------------- // 3.
